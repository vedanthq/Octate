export interface ParsedFile {
  file: string;
  language: 'typescript' | 'javascript' | 'python';
  tree: unknown; // TreeSitter.Tree (avoid importing Tree-sitter types in consumers)
  symbols: Symbol[];
  parseTimeMs: number;
}

export type SymbolKind =
  | 'function' | 'method' | 'class' | 'interface' | 'type'
  | 'constant' | 'variable' | 'module' | 'export' | 'import';

export interface Symbol {
  id: string;           // SHA256(filePath + name + kind) — per D-04
  name: string;
  kind: SymbolKind;
  language: string;
  file: string;
  range: {
    startLine: number;
    endLine: number;
    startColumn: number;
    endColumn: number;
  };
  parentID?: string;    // Direct parent reference — per D-06
  exported: boolean;
  references: string[];
}

export interface ParseOptions {

  signal?: AbortSignal;
}

export interface ParseResult {
  files: ParsedFile[];
  errors: Array<{ file: string; error: Error }>;
  totalTimeMs: number;
}

**2. Create `src/analysis/symbols/index.ts`** — Symbol extraction:

- `symbolId(filePath: string, name: string, kind: SymbolKind): string` — SHA256 hash per D-04:
  ```typescript
  import { createHash } from 'crypto';
  export function symbolId(filePath: string, name: string, kind: SymbolKind): string {
    return createHash('sha256').update(`${filePath}${name}${kind}`).digest('hex');
  }
  ```

- `extractSymbols(parsedFile: ParsedFile): Symbol[]` — Main extraction function:
  - Get language from parsedFile.language
  - Select appropriate queries
  - For each query type (functions, classes, interfaces, types, variables, imports, exports):
    - Run Tree-sitter query on AST
    - For each match, create Symbol object with:
      - `id`: computed via `symbolId(filePath, name, kind)`
      - `name`: from captured node text
      - `kind`: mapped from query type
      - `language`: from parsedFile.language
      - `file`: from parsedFile.file
      - `range`: from node.startPosition/endPosition (convert to 1-indexed lines)
      - `parentID`: find containing node (class contains method, function contains nested function) — per D-06
      - `exported`: check if node is inside export_statement
      - `references`: initially empty array (populated in Phase 3)
  - Return deduplicated symbol array (same name+kind+file = keep one)

- Handle parent detection: For each symbol, walk up the AST to find the nearest enclosing class or function. Set `parentID` to that symbol's ID.
